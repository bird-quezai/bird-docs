With Install 安装工具

页面导航

with-install 组件安装器

为 Vue 组件添加 install 方法，使其成为一个可安装的插件，支持通过 app.use() 进行全局注册。

基础用法 ​

ts
import { withInstall } from '@bird/utils'
import MyComponent from './MyComponent.vue'

// 为组件添加 install 方法
const MyComponentWithInstall = withInstall(MyComponent)

export default MyComponentWithInstall

全局注册 ​

ts
// main.ts
import { createApp } from 'vue'
import App from './App.vue'
import BiButton from './components/Button'

const app = createApp(App)

// 使用 app.use() 全局注册
app.use(BiButton)

app.mount('#app')

在组件库中使用 ​

ts
// components/Button/index.ts
import Button from './src/Button.vue'
import { withInstall } from '@bird/utils'

export const BiButton = withInstall(Button)

export default BiButton
export * from './src/Button.vue'

// 类型声明
declare module 'vue' {
  export interface GlobalComponents {
    BiButton: typeof BiButton
  }
}

按需使用 ​

vue
<template>
  <bi-button type="primary">按钮</bi-button>
</template>

<script setup lang="ts">
// 即使没有全局注册也可以直接使用
import { BiButton } from '@bird/components'
</script>

API ​

withInstall(component) ​

为 Vue 组件添加 install 方法。

参数类型说明
componentComponent | DefineComponentVue 组件

返回值： SFCWithInstall<T> - 添加了 install 方法的组件

install 方法签名：

ts
install(app: App, options?: any): void
ts
import { withInstall } from '@bird/utils'
import MyComponent from './MyComponent.vue'

const MyComponentWithInstall = withInstall(MyComponent)

// 返回的组件包含 install 方法
// MyComponentWithInstall.install(app)

类型定义 ​

ts
type SFCWithInstall<T> = T & Plugin & { name: string }

interface Plugin {
  install: (app: App, options?: any) => void
}

实际应用示例 ​

组件库入口文件 ​

ts
// components/index.ts
import { App } from 'vue'
import { BiButton } from './Button'
import { BiInput } from './Input'
import { BiTable } from './Table'

// 导出所有组件
export * from './Button'
export * from './Input'
export * from './Table'

// 导出安装函数
export function install(app: App) {
  app.use(BiButton)
  app.use(BiInput)
  app.use(BiTable)
}

// 默认导出
export default {
  install
}

在 main.ts 中使用 ​

ts
// main.ts
import { createApp } from 'vue'
import App from './App.vue'
import BirdComponents from '@bird/components'

const app = createApp(App)

// 安装整个组件库
app.use(BirdComponents)

app.mount('#app')

批量注册组件 ​

ts
// utils/register-components.ts
import { App } from 'vue'
import { withInstall } from '@bird/utils'

// 批量注册组件
export function registerComponents(app: App, components: Record<string, any>) {
  Object.entries(components).forEach(([name, component]) => {
    const ComponentWithInstall = withInstall(component)
    app.component(name, ComponentWithInstall)
  })
}

// 使用
import * as Components from './components'
registerComponents(app, Components)

注意事项 ​

  1. 组件必须有 name：组件必须设置 name 属性，否则会抛出错误
  2. 全局注册时机：install 方法会在 app.use() 时自动调用
  3. 组件命名：建议组件名使用 PascalCase 格式，如 BiButton
  4. 类型声明：组件库应该提供 TypeScript 类型声明

Button 组件 ​

vue
<!-- components/Button/src/Button.vue -->
<template>
  <button :class="['bi-button', `bi-button--${type}`]">
    <slot />
  </button>
</template>

<script setup lang="ts">
defineProps<{
  type?: 'primary' | 'success' | 'warning' | 'danger'
}>()
</script>

<script lang="ts">
export default {
  name: 'BiButton'
}
</script>
ts
// components/Button/index.ts
import Button from './src/Button.vue'
import { withInstall } from '@bird/utils'

export const BiButton = withInstall(Button)
export default BiButton

declare module 'vue' {
  export interface GlobalComponents {
    BiButton: typeof BiButton
  }
}

使用方式 ​

ts
// main.ts
import { createApp } from 'vue'
import App from './App.vue'
import { BiButton } from '@bird/components/button'

const app = createApp(App)

// 方式 1：全局注册
app.use(BiButton)

// 方式 2：按需引入（在需要的组件中导入）
// import { BiButton } from '@bird/components/button'

app.mount('#app')
